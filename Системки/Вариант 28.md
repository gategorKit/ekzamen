### Написать программу, которая находит в строке соседей заданного символа и меняет их местами
### Код программы:
``` asm 




MOV @str, R3     	; R3 := указатель на начало строки

MOV @symbol, R1  	; R1 := указатель на символ

MOVB (R1), R1    	; R1 := символ для поиска (загружаем сам символ)



next_char:

MOVB (R3), R0    	; R0 := текущий символ строки

CMPB 24, R0      	; Сравниваем с '$' (код 36 в шестнадцатеричной)

JZ end           	; Если конец строки - завершаем

CMPB R0, R1      	; Сравниваем с искомым символом

JZ found         	; Если нашли - обрабатываем

ADD 1, R3        	; Переходим к следующему символу

JMP next_char



found:

; Проверяем, не первый ли это символ

MOV @str, R2     	; R2 := начало строки

CMP R2, R3       	; Сравниваем позицию с началом строки

JZ first_char    	; Если это первый символ - особый случай



; Проверяем, не последний ли символ (следующий - '$')

MOV R3, R2

ADD 1, R2

MOVB (R2), R0

CMPB 24, R0      	; '$'

JZ last_char     	; Если последний, обрабатываем его отдельно



; Обычный случай (символ в середине строки)

; Меняем местами левого и правого соседей



; Левый сосед (R3 - 1)

MOV R3, R2

SUB 1, R2

MOVB (R2), R0    	; R0 = левый сосед

      

; Правый сосед (R3 + 1)

MOV R3, R2

ADD 1, R2

MOVB (R2), R1    	; R1 = правый сосед

      

; Меняем их местами

MOV R3, R2

SUB 1, R2

MOVB R1, (R2)    	; Правый -> на место левого

      

MOV R3, R2

ADD 1, R2

MOVB R0, (R2)    	; Левый -> на место правого



JMP end



first_char:

; Особый случай: первый символ

; Меняем местами только первый и второй символ



MOVB (R3), R0    	; R0 = первый символ ('a')

MOV R3, R2

ADD 1, R2

MOVB (R2), R1    	; R1 = второй символ ('b')



MOVB R1, (R3)    	; Второй -> на место первого

MOVB R0, (R2)    	; Первый -> на место второго



JMP end



last_char:

; Особый случай: последний символ

; Меняем местами символ перед последним и последний



MOV R3, R2

SUB 1, R2

MOVB (R2), R0    	; Считываем предпоследний символ

MOVB (R3), R1    	; Считываем последний символ



MOVB R1, (R2)    	; Записываем последний на место предпоследнего

MOVB R0, (R3)    	; Записываем предпоследний на место последнего



JMP end



skip:

; Ничего не делаем (символ последний или один)

JMP end



end:

STOP



; Данные

symbol:

DATA "f"  	; Искомый символ



str:

DATA "abcdef$" ; Строка для обработки


```
