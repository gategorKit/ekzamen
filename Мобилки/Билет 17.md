- Помести файл в папку app/src/main/res/drawable.


#### Оценка 3:


- Реализован выбор товаров через CheckBox и ввод количества.

- Фиксированные цены заданы в интерфейсе (можно изменить в коде, если не нужна оценка 5).

- Суммарная стоимость выводится в textViewResult.

- Интерфейс соответствует скриншоту, компоненты имеют осмысленные ID.


#### Оценка 4:


- Текстовые надписи вынесены в strings.xml.

- Обработка ошибок:

- Пустые строки: "Укажите количество для выбранных товаров!".

- Некорректные числа: "Введите корректное число!".

- Отрицательные значения: "Количество не может быть отрицательным!".

- Результат выводится в AlertDialog с детализацией стоимости каждого товара.


#### Оценка 5:


- Добавлены поля для ввода цен (editTextApplePrice, и т.д.).

- Рассчитывается НДС 20% и добавляется к итоговой стоимости.

- В AlertDialog установлена иконка (ic_check).


#### Дополнительные функции:

- Форматирование результата: используется String.format для отображения цен с двумя знаками после запятой.

### MainActivity.java:


``` java
package com.example.checklist;

import android.os.Bundle;
import android.view.View;
import android.widget.Button;
import android.widget.CheckBox;
import android.widget.EditText;
import android.widget.TextView;
import androidx.appcompat.app.AlertDialog;
import androidx.appcompat.app.AppCompatActivity;

public class MainActivity extends AppCompatActivity {

	private CheckBox checkBoxApple, checkBoxStrawberry, checkBoxBlueberry, checkBoxPotatoes;
	private EditText editTextAppleQuantity, editTextStrawberryQuantity, editTextBlueberryQuantity, editTextPotatoesQuantity;
	private EditText editTextApplePrice, editTextStrawberryPrice, editTextBlueberryPrice, editTextPotatoesPrice;
	private TextView textViewResult;

    @Override
	protected void onCreate(Bundle savedInstanceState) {
    	super.onCreate(savedInstanceState);
    	setContentView(R.layout.activity_main);

    	// Инициализация компонентов
    	checkBoxApple = findViewById(R.id.checkBoxApple);
    	checkBoxStrawberry = findViewById(R.id.checkBoxStrawberry);
    	checkBoxBlueberry = findViewById(R.id.checkBoxBlueberry);
    	checkBoxPotatoes = findViewById(R.id.checkBoxPotatoes);

    	editTextAppleQuantity = findViewById(R.id.editTextAppleQuantity);
    	editTextStrawberryQuantity = findViewById(R.id.editTextStrawberryQuantity);
    	editTextBlueberryQuantity = findViewById(R.id.editTextBlueberryQuantity);
    	editTextPotatoesQuantity = findViewById(R.id.editTextPotatoesQuantity);

    	editTextApplePrice = findViewById(R.id.editTextApplePrice);
    	editTextStrawberryPrice = findViewById(R.id.editTextStrawberryPrice);
    	editTextBlueberryPrice = findViewById(R.id.editTextBlueberryPrice);
    	editTextPotatoesPrice = findViewById(R.id.editTextPotatoesPrice);

    	textViewResult = findViewById(R.id.textViewResult);

    	Button buttonCalculate = findViewById(R.id.buttonCalculate);
    	buttonCalculate.setOnClickListener(new View.OnClickListener() {
        	@Override
        	public void onClick(View v) {
            	calculateCheck();
        	}
    	});
	}

	private void calculateCheck() {
    	StringBuilder result = new StringBuilder();
    	double total = 0.0;

    	// Проверка и расчет для Apple
    	if (checkBoxApple.isChecked()) {
        	String quantityText = editTextAppleQuantity.getText().toString();
        	String priceText = editTextApplePrice.getText().toString();

        	if (quantityText.isEmpty()) {
            	textViewResult.setText(getString(R.string.errorEmpty));
            	return;
        	}
        	if (priceText.isEmpty()) {
            	textViewResult.setText(getString(R.string.errorPriceEmpty));
            	return;
        	}

        	try {
            	double quantity = Double.parseDouble(quantityText);
            	double price = Double.parseDouble(priceText);

            	if (quantity < 0) {
                	textViewResult.setText(getString(R.string.errorNegative));
                	return;
            	}
            	if (price < 0) {
                	textViewResult.setText(getString(R.string.errorPriceNegative));
                	return;
            	}

            	double cost = quantity * price;
            	total += cost;
            	result.append(String.format("%s: %.1f x %.2f = %.2f\n", getString(R.string.itemApple), quantity, price, cost));
        	} catch (NumberFormatException e) {
            	textViewResult.setText(getString(R.string.errorInvalid));
            	return;
        	}
    	}

    	// Проверка и расчет для Strawberry
    	if (checkBoxStrawberry.isChecked()) {
        	String quantityText = editTextStrawberryQuantity.getText().toString();
        	String priceText = editTextStrawberryPrice.getText().toString();

        	if (quantityText.isEmpty()) {
            	textViewResult.setText(getString(R.string.errorEmpty));
            	return;
        	}
        	if (priceText.isEmpty()) {
            	textViewResult.setText(getString(R.string.errorPriceEmpty));
            	return;
        	}

        	try {
            	double quantity = Double.parseDouble(quantityText);
            	double price = Double.parseDouble(priceText);

            	if (quantity < 0) {
                	textViewResult.setText(getString(R.string.errorNegative));
                	return;
            	}
            	if (price < 0) {
                	textViewResult.setText(getString(R.string.errorPriceNegative));
                	return;
            	}

            	double cost = quantity * price;
            	total += cost;
            	result.append(String.format("%s: %.1f x %.2f = %.2f\n", getString(R.string.itemStrawberry), quantity, price, cost));
        	} catch (NumberFormatException e) {
            	textViewResult.setText(getString(R.string.errorInvalid));
            	return;
        	}
    	}

    	// Проверка и расчет для Blueberry
    	if (checkBoxBlueberry.isChecked()) {
        	String quantityText = editTextBlueberryQuantity.getText().toString();
        	String priceText = editTextBlueberryPrice.getText().toString();

        	if (quantityText.isEmpty()) {
            	textViewResult.setText(getString(R.string.errorEmpty));
            	return;
        	}
        	if (priceText.isEmpty()) {
            	textViewResult.setText(getString(R.string.errorPriceEmpty));
            	return;
        	}

        	try {
            	double quantity = Double.parseDouble(quantityText);
            	double price = Double.parseDouble(priceText);

            	if (quantity < 0) {
                	textViewResult.setText(getString(R.string.errorNegative));
                	return;
            	}
            	if (price < 0) {
                	textViewResult.setText(getString(R.string.errorPriceNegative));
                	return;
            	}

            	double cost = quantity * price;
            	total += cost;
            	result.append(String.format("%s: %.1f x %.2f = %.2f\n", getString(R.string.itemBlueberry), quantity, price, cost));
        	} catch (NumberFormatException e) {
            	textViewResult.setText(getString(R.string.errorInvalid));
            	return;
        	}
    	}

    	// Проверка и расчет для Potatoes
    	if (checkBoxPotatoes.isChecked()) {
        	String quantityText = editTextPotatoesQuantity.getText().toString();
        	String priceText = editTextPotatoesPrice.getText().toString();

        	if (quantityText.isEmpty()) {
            	textViewResult.setText(getString(R.string.errorEmpty));
            	return;
        	}
        	if (priceText.isEmpty()) {
            	textViewResult.setText(getString(R.string.errorPriceEmpty));
            	return;
        	}

        	try {
            	double quantity = Double.parseDouble(quantityText);
            	double price = Double.parseDouble(priceText);

            	if (quantity < 0) {
                	textViewResult.setText(getString(R.string.errorNegative));
                	return;
            	}
            	if (price < 0) {
                	textViewResult.setText(getString(R.string.errorPriceNegative));
                	return;
            	}

            	double cost = quantity * price;
            	total += cost;
            	result.append(String.format("%s: %.1f x %.2f = %.2f\n", getString(R.string.itemPotatoes), quantity, price, cost));
        	} catch (NumberFormatException e) {
            	textViewResult.setText(getString(R.string.errorInvalid));
            	return;
        	}
    	}

    	// Если ничего не выбрано
    	if (!checkBoxApple.isChecked() && !checkBoxStrawberry.isChecked() &&
            	!checkBoxBlueberry.isChecked() && !checkBoxPotatoes.isChecked()) {
        	textViewResult.setText("Выберите хотя бы один товар!");
        	return;
    	}

    	// Расчет НДС (20%)
    	double vat = total * 0.20;
    	double totalWithVat = total + vat;

    	// Формирование итогового чека
    	result.append(String.format("\nSubtotal: %.2f\n", total));
    	result.append(String.format("VAT (20%%): %.2f\n", vat));
    	result.append(String.format("Total: %.2f", totalWithVat));

    	// Для оценки 3: вывод только суммы в надпись
    	textViewResult.setText(String.format("Total: %.2f", total));

    	// Для оценки 4 и 5: вывод в диалоговое окно
    	AlertDialog.Builder builder = new AlertDialog.Builder(this);
    	builder.setTitle(getString(R.string.dialogTitle))
            	.setIcon(R.drawable.ic_check) // Иконка для оценки 5
            	.setMessage(result.toString())
            	.setPositiveButton("OK", null)
            	.show();
	}
}
```
