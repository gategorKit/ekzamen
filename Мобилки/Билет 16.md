- Реализована конвертация из метров в мм, см, м, км
- Интерфейс соответствует скриншоту
- Компоненты имеют осмысленные ID: `txtFrom`, `labTo`, `spinnerFrom`, `spinnerTo`, `buttonConvert`

### Оценка 4:
- Текстовые надписи вынесены в `strings.xml`
- Обработка ошибок:
  - Пустое поле: "Поле не может быть пустым!"
  - Некорректное число: "Введите корректное число!"
  - Отрицательное значение: "Отрицательные значения не допускаются!"
- Добавлен второй Spinner (`spinnerFrom`) для выбора исходной единицы измерения
- Ошибки выводятся в поле результата (`labTo`)

---

### Оценка 5:
- Добавлена конвертация единиц информации: бит, байт, КБ, МБ, ГБ
- Сохранена конвертация единиц длины
- Для переключения между режимами (длина/информация) используется флаг `isLengthMode`:
  ```java
  // Для тестирования можно менять значение флага
  isLengthMode = true;  // режим длины
  isLengthMode = false; // режим информации
  ``` 


#### Дополнительные функции:

- Форматирование результата: используется String.format("%.2f", result) для отображения двух знаков после запятой.

### MainActivity.java


``` java
package com.example.unitconverter;

import android.os.Bundle;
import android.view.View;
import android.widget.ArrayAdapter;
import android.widget.Button;
import android.widget.EditText;
import android.widget.Spinner;
import android.widget.TextView;
import androidx.appcompat.app.AppCompatActivity;

public class MainActivity extends AppCompatActivity {

	private EditText txtFrom;
	private TextView labTo;
	private Spinner spinnerFrom, spinnerTo;
	private boolean isLengthMode = false; // Флаг для режима: длина или информация

	@Override
	protected void onCreate(Bundle savedInstanceState) {
    	super.onCreate(savedInstanceState);
    	setContentView(R.layout.activity_main);

    	// Инициализация компонентов
    	txtFrom = findViewById(R.id.txtFrom);
    	labTo = findViewById(R.id.labTo);
    	spinnerFrom = findViewById(R.id.spinnerFrom);
    	spinnerTo = findViewById(R.id.spinnerTo);
    	Button buttonConvert = findViewById(R.id.buttonConvert);

    	// Изначально режим длины
    	updateSpinners();

    	// Обработчик кнопки конвертации
    	buttonConvert.setOnClickListener(new View.OnClickListener() {
        	@Override
        	public void onClick(View v) {
            	performConversion();
        	}
    	});
	}

	// Обновление списков в зависимости от режима (длина или информация)
	private void updateSpinners() {
    	ArrayAdapter<CharSequence> adapter;
    	if (isLengthMode) {
        	adapter = ArrayAdapter.createFromResource(this,
                	R.array.length_units, android.R.layout.simple_spinner_item);
    	} else {
        	adapter = ArrayAdapter.createFromResource(this,
                	R.array.data_units, android.R.layout.simple_spinner_item);
    	}
    	adapter.setDropDownViewResource(android.R.layout.simple_spinner_dropdown_item);
    	spinnerFrom.setAdapter(adapter);
    	spinnerTo.setAdapter(adapter);
	}

	// Выполнение конвертации
	private void performConversion() {
    	String inputText = txtFrom.getText().toString();

    	// Проверка на пустое поле
    	if (inputText.isEmpty()) {
        	labTo.setText(getString(R.string.errorEmpty));
        	return;
    	}

    	try {
        	double value = Double.parseDouble(inputText);

        	// Проверка на отрицательное значение
        	if (value < 0) {
            	labTo.setText(getString(R.string.errorNegative));
            	return;
        	}

        	double result;
        	if (isLengthMode) {
            	// Конвертация длины
            	result = convertLength(value, spinnerFrom.getSelectedItem().toString(),
                    	spinnerTo.getSelectedItem().toString());
        	} else {
            	// Конвертация информации
            	result = convertData(value, spinnerFrom.getSelectedItem().toString(),
                    	spinnerTo.getSelectedItem().toString());
        	}
        	labTo.setText(String.format("%.2f", result));
    	} catch (NumberFormatException e) {
        	labTo.setText(getString(R.string.errorInvalid));
    	}
	}

	// Конвертация единиц длины
	private double convertLength(double value, String fromUnit, String toUnit) {
    	// Приведение к метрам
    	double inMeters;
    	switch (fromUnit) {
        	case "mm":
            	inMeters = value / 1000;
            	break;
        	case "cm":
            	inMeters = value / 100;
            	break;
        	case "km":
            	inMeters = value * 1000;
            	break;
        	case "m":
        	default:
            	inMeters = value;
            	break;
    	}

    	// Конвертация из метров в целевую единицу
    	switch (toUnit) {
        	case "mm":
            	return inMeters * 1000;
        	case "cm":
            	return inMeters * 100;
        	case "km":
            	return inMeters / 1000;
        	case "m":
        	default:
            	return inMeters;
    	}
	}

	// Конвертация единиц информации
	private double convertData(double value, String fromUnit, String toUnit) {
    	// Приведение к битам
    	double inBits;
    	switch (fromUnit) {
        	case "byte":
            	inBits = value * 8;
            	break;
        	case "KB":
            	inBits = value * 8 * 1024;
            	break;
        	case "MB":
            	inBits = value * 8 * 1024 * 1024;
            	break;
        	case "GB":
            	inBits = value * 8 * 1024 * 1024 * 1024;
            	break;
        	case "bit":
        	default:
            	inBits = value;
            	break;
    	}

    	// Конвертация из битов в целевую единицу
    	switch (toUnit) {
        	case "byte":
            	return inBits / 8;
        	case "KB":
            	return inBits / (8 * 1024);
        	case "MB":
            	return inBits / (8 * 1024 * 1024);
        	case "GB":
            	return inBits / (8 * 1024 * 1024 * 1024);
        	case "bit":
        	default:
            	return inBits;
    	}
	}

	// Метод для переключения режима (для тестирования можно добавить кнопку)
	public void toggleMode(View view) {
    	isLengthMode = !isLengthMode;
    	updateSpinners();
    	labTo.setText("--");
    	txtFrom.setText("");
	}
}
``` 
